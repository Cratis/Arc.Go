type Detail struct { ID concepts.UUID `json:"id"`; Label string `json:"label"` }
//arc:command response=none
//arc:authorize roles=Editor,Admin,Editor
// Register returns a server-consumed value, NOT a client Detail response.
type Register struct {
 ID concepts.UUID `json:"id"`
 Name string `json:"name" rules:"[{\"name\":\"notEmpty\",\"message\":\"Name required\"},{\"name\":\"maxLength\",\"arguments\":[40],\"message\":\"Name too long\"}]"`
 Quantity int `json:"quantity" rules:"[{\"name\":\"greaterThanOrEqual\",\"arguments\":[1],\"message\":\"Quantity must be positive\"}]"`
 Enabled bool `json:"enabled"`
 Description *string `json:"description,omitempty"`
 Tags []string `json:"tags" validate:"required"`
 Exact string `json:"EXACT-name"`
}
func (Register) Handle() (Detail, error) { return Detail{}, nil }
