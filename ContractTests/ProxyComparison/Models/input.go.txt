// Metadata-only generic concepts. Discovery must never execute these codecs.
type Value[T any] struct { value T }
func (v Value[T]) ConceptValue() T { panic("marker executed") }
func (Value[T]) MarshalText() ([]byte,error) { panic("codec executed") }
func (Value[T]) MarshalJSON() ([]byte,error) { panic("codec executed") }
func (*Value[T]) UnmarshalText([]byte) error { panic("codec executed") }
func (*Value[T]) UnmarshalJSON([]byte) error { panic("codec executed") }

//arc:enum members=StateDraft:Draft,StatePublished:Published,StateArchived:Archived,StateAlias:Alias,StateDefault:Default
 type State int32
const (StateDraft State = 0; StatePublished State = 7; StateArchived State = -2; StateAlias State = 7; StateDefault State = 42)
//arc:enum flags=true members=AccessNone:None,AccessRead:Read,AccessWrite:Write
 type Access uint32
const (AccessNone Access = 0; AccessRead Access = 1; AccessWrite Access = 4)

 type Detail struct { ID concepts.UUID `json:"id"`; Label string `json:"label"` }
 type NoticeContract interface { notice() }
//arc:model
//arc:derived interface=NoticeContract
 type Notice struct { Title string `json:"title"` }
func (Notice) notice() {}
//arc:model
//arc:derived interface=NoticeContract base=Notice id=1578F20A-CD63-456F-98AA-C97DAF05D0FA
 type UrgentNotice struct { Notice; Priority int32 `json:"priority"` }
