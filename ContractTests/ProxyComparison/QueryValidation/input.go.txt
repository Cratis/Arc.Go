//arc:namespace Shop.Validation
package consumer

//arc:readmodel
//arc:allow-anonymous
type Match struct { Text string `json:"text"` }

type SearchArguments struct {
 Text *string `json:"EXACT-name" rules:"[{\"name\":\"notEmpty\",\"message\":\"Text is required\"},{\"name\":\"minLength\",\"arguments\":[3],\"message\":\"Text needs three UTF-16 units\"}]"`
 Present *string `json:"present" rules:"[{\"name\":\"notNull\",\"message\":\"Present is required\"}]"`
 Range *string `json:"range" rules:"[{\"name\":\"length\",\"arguments\":[2,4],\"message\":\"Range needs two to four units\"}]"`
 Warning *string `json:"warning" rules:"[{\"name\":\"maxLength\",\"arguments\":[3],\"message\":\"Warning limit is three\",\"severity\":2}]"`
 Info *string `json:"info" rules:"[{\"name\":\"minLength\",\"arguments\":[2],\"message\":\"Info minimum is two\",\"severity\":1}]"`
 Required string `json:"required" query:"required" rules:"[{\"name\":\"maxLength\",\"arguments\":[4],\"message\":\"Required limit is four\"}]"`
}

var Calls int
func (Match) Search(args SearchArguments) ([]Match, error) {
 Calls++
 return []Match{{Text: *args.Text}}, nil
}
